#!/usr/bin/env node
// Emitted normal-game capture and separately labelled interruption inspection.
import {chromium} from 'playwright';
import path from 'node:path';
import http from 'node:http';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import fs from 'node:fs';
const opt=(name,fallback)=>{const i=process.argv.indexOf(name);return i<0?fallback:process.argv[i+1];};
const site=fs.realpathSync(opt('--site','_site')),out=path.resolve(opt('--evidence-dir','.ci-scratch/triple-splashdown/post-impact/raw'));
fs.mkdirSync(out,{recursive:true});
const hash=b=>createHash('sha256').update(b).digest('hex');
const manifestBytes=fs.readFileSync(path.join(site,'inkwave-build.json')),manifest=JSON.parse(manifestBytes);
assert.equal(hash(JSON.stringify(manifest.artifacts)),manifest.contentHash);
for(const [name,digest] of Object.entries(manifest.artifacts))assert.equal(hash(fs.readFileSync(path.join(site,name))),digest,name);
const served=new Set(),mime={'.html':'text/html','.js':'text/javascript','.mjs':'text/javascript','.json':'application/json','.css':'text/css','.png':'image/png','.svg':'image/svg+xml','.woff2':'font/woff2','.glb':'model/gltf-binary','.webp':'image/webp','.jpg':'image/jpeg','.mp3':'audio/mpeg'};
const server=http.createServer((req,res)=>{try{
 const name=decodeURIComponent(new URL(req.url,'http://local').pathname),rel=name==='/'?'index.html':name.slice(1),file=path.resolve(site,rel);
 assert.ok(file.startsWith(site+'/')&&fs.statSync(file).isFile());const body=fs.readFileSync(file);
 assert.equal(hash(body),rel==='inkwave-build.json'?hash(manifestBytes):manifest.artifacts[rel],rel+' served build identity');served.add(rel);
 res.writeHead(200,{'content-type':mime[path.extname(file)]||'application/octet-stream','cache-control':'no-store'});res.end(body);
}catch{res.writeHead(404);res.end();}});
await new Promise(r=>server.listen(0,'127.0.0.1',r));
const browser=await chromium.launch({...(process.env.CHROMIUM_PATH?{executablePath:process.env.CHROMIUM_PATH}:{}),headless:true});
const page=await browser.newPage({viewport:{width:1280,height:800}}),errors=[];
page.on('pageerror',e=>errors.push(e.message));page.on('console',m=>{if(m.type()==='error')errors.push(m.text().slice(0,400));});
await page.addInitScript(()=>{localStorage.setItem('inkwave.settings',JSON.stringify({quality:'low',shadows:false,bloom:false,minimap:false}));let seed=12345;Math.random=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296;};});
try{
 await page.goto('http://127.0.0.1:'+server.address().port+'/?autostart=180',{timeout:120000});
 await page.waitForFunction(()=>window.__G?.mode==='match'&&window.__inkwave?.match?.local&&!window.__inkwave.match.attract&&window.__inkwave.match.state==='playing',null,{timeout:180000});
 await page.waitForTimeout(800);
 const info=await page.evaluate(async()=>{
  const g=window.__inkwave,G=window.__G;g.debug.freeze();g.debug.freezeBots();
  const manifest=await fetch('/inkwave-build.json').then(r=>r.json()),prefix='/_versions/'+manifest.build.revision+'/';
  const api={...await import(prefix+'src/config.js'),...await import(prefix+'src/core/ctx.js')};
  const {applyCatalogueChoice,CATALOGUE_STORAGE}=await import(prefix+'patches/splatoon3/runtime/catalogue.mjs');
  const a=g.match.local;window.splashProbe={g,G,a,api,events:[],origins:{},frame:0,rows:[]};
  if(!a)throw Error('No local actor');
  // Leave the opening spawner through the ordinary fire input, then finish
  // its native flight before measuring a ground activation.
  g.debug.step(1000/60);g.debug.fire(true);g.debug.step(1000/60);g.debug.fire(false);
  for(let i=0;i<12&&a.s3?.squidSpawn;i++)g.debug.step(20*1000/60);
  if(a.s3?.squidSpawn||!a.grounded)throw Error('Opening spawn has not landed');
  G.fx.clear();G.projectiles.clear();
  const choice={sub:'bomb',special:'tripleSplashdown'};
  localStorage.setItem(CATALOGUE_STORAGE,JSON.stringify({[a.weaponId]:choice}));
  if(!applyCatalogueChoice(a,choice,api))throw Error('Catalogue choice rejected');a.special=a.specialCost();a.invuln=0;
  api.on('all:special',e=>{
    if(e.actor!==a||e.id!=='tripleSplashdown')return;
    const p=window.splashProbe,event={frame:p.frame,time:G.time,action:e.action,index:e.index??-1,pos:e.pos,
      cameraAtEvent:{position:G.camera.position.toArray(),quaternion:G.camera.quaternion.toArray(),fov:G.camera.fov}};
    p.events.push(event);
    if(e.action==='impact'){
      const group=(e.index??-1)<0?'central':'paired';
      const origin=p.origins[group]??={frame:p.frame,time:G.time,indices:[]};
      if(origin.frame!==p.frame||origin.time!==G.time)throw Error('Paired impacts are not simultaneous in this capture');
      origin.indices.push(e.index??-1);
    }
  });
  const gl=G.renderer.getContext(),debug=gl.getExtension('WEBGL_debug_renderer_info');
  return {revision:manifest.build.revision,contentHash:manifest.contentHash,inputHash:manifest.inputHash,map:g.mapDef.id,pos:a.pos.toArray(),webgl2:gl instanceof WebGL2RenderingContext,renderer:debug?gl.getParameter(debug.UNMASKED_RENDERER_WEBGL):gl.getParameter(gl.RENDERER),browser:navigator.userAgent};
 });
 console.log('Normal game ready; preserving existing game camera. Capturing native impact offsets.');
 const captures=[];
 await page.evaluate(()=>window.splashProbe.g.debug.key('KeyQ',true));
 for(let frame=1;frame<=180;frame++){
  const row=await page.evaluate(frame=>{
    const p=window.splashProbe,{g,G,a}=p;p.frame=frame;g.debug.step(1000/60);g.debug.key('KeyQ',false);
    const camera={kind:'existing game camera',position:G.camera.position.toArray(),quaternion:G.camera.quaternion.toArray(),
      fov:G.camera.fov,aspect:G.camera.aspect,near:G.camera.near,far:G.camera.far,zoom:G.camera.zoom,
      matrixWorld:G.camera.matrixWorld.toArray(),projectionMatrix:G.camera.projectionMatrix.toArray(),rigYaw:g.rig.yaw,rigPitch:g.rig.pitch};
    const wanted=[];
    for(const [group,origin]of Object.entries(p.origins)){
      const offset=frame-origin.frame;if([0,2,4,6,10].includes(offset))wanted.push({group,offset,impactFrame:origin.frame,impactTime:origin.time});
    }
    const row={frame,time:G.time,actorPos:a.pos.toArray(),phase:a.specialActive?.phase??null,fists:(G.projectiles._catalogueFists||[]).length,camera,wanted,
      drawCalls:G.renderer.info.render.calls,triangles:G.renderer.info.render.triangles};p.rows.push(row);
    return {row,done:!!p.origins.paired&&frame>=p.origins.paired.frame+10};
  },frame);
  for(const wanted of row.row.wanted){
    const name=`normal-${wanted.group}-plus-${String(wanted.offset).padStart(2,'0')}-sim-f${String(frame).padStart(3,'0')}.png`;
    const file=path.join(out,name);await page.screenshot({path:file});
    captures.push({...wanted,frame,time:row.row.time,camera:row.row.camera,file:name,sha256:hash(fs.readFileSync(file))});
    console.log(JSON.stringify({capture:name,time:row.row.time,impactTime:wanted.impactTime,cameraPosition:row.row.camera.position}));
  }
  if(row.done)break;
 }
 const result=await page.evaluate(()=>{const {events,origins,rows,G}=window.splashProbe;return {events,origins,rows,vfxErrors:G.fx.s3SpecialVfx.errors};});
 const receipt={scope:'Normal production game camera on tidewater, native opening launch and Q activation. Fixed native debug stepping 1/60; one ordinary game render per simulation step, selected still screenshots only. Bots frozen; catalogue/gauge setup only. No fixed inspection camera, physics or VFX modification.',
   implementationCommit:'8437516cebd527e033711631174c4a711083aa26',simulationHz:60,dt:1/60,captureFPS:null,encodedFPS:null,
   referenceRecording:'Parent reports encoded 25fps; not inspected in this task; recording frames are not simulation frames.',viewport:{width:1280,height:800},
   qualitySetup:{quality:'low',shadows:false,bloom:false,minimap:false},...info,...result,captures,errors,served:[...served]};
 fs.writeFileSync(path.join(out,'receipt.json'),JSON.stringify(receipt,null,2)+'\n');
 assert.deepEqual(errors,[]);assert.equal(info.webgl2,true);assert.equal(result.vfxErrors,0);assert.equal(captures.length,10);
 assert.deepEqual(result.events.filter(e=>e.action==='impact').map(e=>e.index),[-1,1,0]);
 for(const group of ['central','paired'])assert.deepEqual(captures.filter(c=>c.group===group).map(c=>c.offset),[0,2,4,6,10]);
 console.log(JSON.stringify({out,origins:result.origins,captures:captures.length,errors}));
}catch(e){console.log(JSON.stringify({failure:String(e),errors}));throw e;}finally{await browser.close();await new Promise(r=>server.close(r));}
