from pathlib import Path
from PIL import Image,ImageDraw,ImageFont
import json,hashlib
raw=Path('.ci-scratch/triple-splashdown/post-impact/raw')
out=Path('reports/inkwave-triple-splashdown-post-impact-2026-10-23')
r=json.loads((raw/'receipt.json').read_text())
assert r['implementationCommit']=='8437516cebd527e033711631174c4a711083aa26'
assert r['simulationHz']==60 and not r['errors'] and r['vfxErrors']==0
assert len(r['captures'])==10
out.mkdir(exist_ok=True)
font='/usr/share/fonts/truetype/open-sans/OpenSans-Regular.ttf'
bold='/usr/share/fonts/truetype/open-sans/OpenSans-Semibold.ttf'
fonts={k:ImageFont.truetype(p,s) for k,p,s in [('title',bold,30),('group',bold,23),('caption',bold,20),('small',font,17)]}
margin,gap,cellw,cellh=24,12,512,320
width=margin*2+cellw*5+gap*4
rowheight=420;height=112+2*rowheight+72
sheet=Image.new('RGB',(width,height),'#101621');d=ImageDraw.Draw(sheet)
d.text((margin,16),'Triple Splashdown | existing normal game camera | frozen implementation 8437516',font=fonts['title'],fill='#f2f5fa')
d.text((margin,62),'Native impact offsets: +0, +2, +4, +6, +10 simulation ticks. Full viewport images; proportional reduction only.',font=fonts['small'],fill='#aebace')
for row,group in enumerate(['central','paired']):
 y=112+row*rowheight;origin=r['origins'][group]
 label='Central player impact' if group=='central' else 'Paired fist impacts (indices 1 and 0, same tick)'
 d.text((margin,y),f"{label} | native sim f{origin['frame']:03} | event t={origin['time']:.6f}s",font=fonts['group'],fill='#d5e9ff')
 captures=[c for c in r['captures'] if c['group']==group]
 assert [c['offset'] for c in captures]==[0,2,4,6,10]
 for col,c in enumerate(captures):
  p=raw/c['file'];assert hashlib.sha256(p.read_bytes()).hexdigest()==c['sha256']
  assert c['frame']==origin['frame']+c['offset']
  assert abs((c['time']-origin['time'])-c['offset']/60)<1e-8
  im=Image.open(p).convert('RGB');assert im.size==(1280,800)
  x=margin+col*(cellw+gap)
  sheet.paste(im.resize((cellw,cellh),Image.Resampling.LANCZOS),(x,y+36))
  d.text((x,y+365),f"+{c['offset']} ticks | sim f{c['frame']:03} | t={c['time']:.6f}s",font=fonts['caption'],fill='#f2f5fa')
  cam=c['camera']['position']
  d.text((x,y+393),f"cam ({cam[0]:.2f}, {cam[1]:.2f}, {cam[2]:.2f}) | FOV {c['camera']['fov']:.1f}",font=fonts['small'],fill='#aebace')
d.text((margin,height-55),'60 Hz fixed simulation; one native game render per step. Selected stills only: no continuous capture FPS or encoded video FPS.',font=fonts['small'],fill='#c1cede')
d.text((margin,height-29),'No retail footage. Reference recording frames (parent: encoded 25 fps) are not mapped to these simulation ticks. Camera matrices: receipt.json.',font=fonts['small'],fill='#c1cede')
sheet.save(out/'normal-game-camera.png',optimize=True)
# Publish the actual receipt, with full per-tick cameras, not just sheet captions.
(out/'receipt.json').write_text(json.dumps(r,indent=2)+'\n')
(out/'capture.log').write_bytes(Path('.ci-scratch/triple-splashdown/post-impact/capture.log').read_bytes())
print(json.dumps({'sheet':str(out/'normal-game-camera.png'),'size':sheet.size,'bytes':(out/'normal-game-camera.png').stat().st_size,'central':r['origins']['central'],'paired':r['origins']['paired'],'frames':[c['frame'] for c in r['captures']]}))
