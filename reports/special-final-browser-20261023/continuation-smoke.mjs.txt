const base='/workspace/inkwave-special-candidate/.ci-scratch/special-final-browser',source=base+'/site-after',out=base+'/continuation-clean';fs.mkdirSync(out,{recursive:true});
// Reuse the exact registered production-menu fixture HTML, selecting its already
// built styles/runtime. No production or runner input is changed.
const text=fs.readFileSync('/workspace/inkwave-special-candidate/scripts/check-inkwave-responsive.mjs','utf8');
const patchStyles='<link rel="stylesheet" href="/patches/splatoon3/ui.css">';
const runtimeInstaller=text.match(/const runtimeInstaller = patchStyles \? `([\s\S]*?)` : '';/)[1];
const template=text.match(/const html = `([\s\S]*?)`;\nconst server =/)[1];

try{

 const entry={name,viewport,errors:[],consoleErrors:[],httpErrors:[],operations:[],captures:[]};report.cases.push(entry);save();
 page.on('response',r=>{if(r.status()>=400)entry.httpErrors.push({url:r.url(),status:r.status()});});page.on('pageerror',e=>entry.errors.push(e.stack));page.on('console',m=>{if(m.type()==='error')entry.consoleErrors.push(m.text());});
 try{
 await page.goto(`http://127.0.0.1:${server.address().port}/__menus?netmock=1&mockauto=0&mocklat=0&news=0`);

 const settle=(page,observe=false)=>page.evaluate(settleResponsiveAnimationsInPage,observe);
 const capture=async(label,selector)=>{
  if(label==='loadout-continue')await page.waitForFunction(responsiveAnimationsSettled);else await settle(page);
  const geometry=await page.locator(selector).evaluateAll(els=>els.filter(el=>el.getClientRects().length&&!el.closest('.is-leaving')).map(el=>{let r=el.getBoundingClientRect();return {width:r.width,height:r.height,left:r.left,right:r.right,viewport:innerWidth};}));
  assert(geometry.length);for(const r of geometry){assert(r.left>=-1&&r.right<=r.viewport+1,label+': overflow');assert(r.width>=43&&r.height>=43,label+': touch size');}
  const image=name+'-'+label+'.png';await page.screenshot({path:path.join(out,image)});entry.captures.push({label,image,geometry});save();
 };
 await checkResultContinuation({page,tap,settle,capture,entry,onOperation:(id,boundary)=>{entry.operations.push({id,boundary,wall:Date.now()});save();}});
 assert.deepEqual(entry.errors,[]);assert.deepEqual(entry.consoleErrors,[]);entry.status='passed';console.log(name+' continuation passed');
 }catch(e){entry.status='failed';entry.failure=e.stack;console.error(e.stack);await page.screenshot({path:path.join(out,name+'-failure.png')}).catch(()=>{});process.exitCode=1;}finally{save();await context.close();}
}
}finally{save();await browser.close();await new Promise(r=>server.close(r));}
