def maxdiff(v,w):
    if isinstance(v,dict):
        assert v.keys()==w.keys()
        return max((maxdiff(v[k],w[k]) for k in v),default=0)
    if isinstance(v,list):
        assert len(v)==len(w)
        return max((maxdiff(p,q) for p,q in zip(v,w)),default=0)
    if isinstance(v,(float,int)):
        assert math.isfinite(v) and math.isfinite(w)
        return abs(v-w)
    assert v==w
    return 0
camera=maxdiff(x['camera'],y['camera']);character=maxdiff(x['characterRoot'],y['characterRoot']);bones=maxdiff(x['bones'],y['bones'])
assert camera<1e-8,('camera mismatch',camera)
assert character<1e-8,('character root mismatch',character)
assert bones<1e-8,('character bone mismatch',bones)
assert len(x['matrices'])==len(y['matrices'])==4
