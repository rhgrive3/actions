import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import {execFileSync,spawnSync} from 'node:child_process';
const codeRoot=process.cwd(),root='/workspace/projectile-closure-probes',dir=path.join(codeRoot,'reports/projectile-current-pair-closure-20261023');
const sources=JSON.parse(fs.readFileSync(path.join(dir,'old-bypass.json'))).sources;
const originalEvent=JSON.parse(fs.readFileSync(path.join(codeRoot,'reports/projectile-current-pair-route-20261023/pr1857-base726.json')));
const git=(...args)=>execFileSync('git',['-C',root,...args],{encoding:'utf8'}).trim();
const commit=message=>{git('-c','user.name=QA probe','-c','user.email=qa@example.invalid','commit','--quiet','-m',message);return git('rev-parse','HEAD');};
const records=[];
function run(label,relative,tracked,positive=false) {
  const event=structuredClone(originalEvent);event.pull_request.head.sha=git('rev-parse','HEAD');
  const eventFile=path.join(dir,label+'-event.json'),receipt=path.join(dir,label+'-receipt.json');fs.writeFileSync(eventFile,JSON.stringify(event,null,2)+'\n');
  const sentinel='prior receipt must survive closure rejection\n';fs.writeFileSync(receipt,sentinel);
  const result=spawnSync(process.execPath,[codeRoot+'/scripts/verify-inkwave-projectile-current-pair.mjs','--baseline','/workspace/projectile-route-base726','--candidate',root,'--event',eventFile,'--receipt',receipt],{encoding:'utf8',env:{...process.env,GITHUB_EVENT_NAME:'pull_request',GITHUB_REPOSITORY:'rhgrive3/actions'}});
  fs.writeFileSync(path.join(dir,label+'.log'),result.stdout+result.stderr);
  assert.equal(result.signal,null);assert.equal(result.status,positive?0:1,result.stderr);
  if(!positive){assert.equal(fs.readFileSync(receipt,'utf8'),sentinel);assert.match(result.stderr,/complete production inventory|source symlinks rejected/);}
  const record={label,relative,tracked,head:event.pull_request.head.sha,status:result.status,priorReceiptPreserved:!positive,reason:positive?'valid restored pair accepted':result.stderr.split('\n').find(l=>l.includes('AssertionError'))};
  records.push(record);return record;
}
let index=0;
for(const [relative,body]of Object.entries(sources)) {
  const file=path.join(root,relative);assert.equal(fs.existsSync(file),false);fs.mkdirSync(path.dirname(file),{recursive:true});fs.writeFileSync(file,body);
  run('probe-'+index+'-untracked',relative,false);
  git('add','--',relative);commit('audit: track closure probe '+relative);run('probe-'+index+'-tracked',relative,true);
  fs.unlinkSync(file);git('add','--',relative);commit('audit: remove closure probe '+relative);index++;
}
const outside=fs.mkdtempSync('/tmp/projectile-closure-cli-outside-');fs.writeFileSync(path.join(outside,'owned.md'),'owned outside marker\n');
for(const [relative,directory]of [['inkwave-public/tests/escape.md',false],['inkwave-public/nested/tests',true]]) {
  const file=path.join(root,relative);fs.mkdirSync(path.dirname(file),{recursive:true});if(fs.existsSync(file)){assert.equal(fs.readdirSync(file).length,0);fs.rmdirSync(file);}fs.symlinkSync(directory?outside:path.join(outside,'owned.md'),file);
  run('probe-'+index+'-untracked-link',relative,false);git('add','--',relative);commit('audit: track closure symlink '+relative);run('probe-'+index+'-tracked-link',relative,true);
  fs.unlinkSync(file);git('add','--',relative);commit('audit: remove closure symlink '+relative);index++;
}
assert.equal(git('status','--porcelain'),'');run('restored-valid-pair',null,false,true);
fs.writeFileSync(path.join(dir,'real-cli-probes.json'),JSON.stringify(records,null,2)+'\n');
console.log({negativeCases:records.filter(r=>r.status===1).length,validRestoredPair:records.at(-1),allNegativeReceiptsPreserved:records.filter(r=>r.status===1).every(r=>r.priorReceiptPreserved)});
